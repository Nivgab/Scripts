getallenvironments = function()
local scripts = {}
table.foreach(getreg(), function(x, y)
if type(y) == 'thread' then
local env = gettenv(y)
if env and not scripts[env] then
scripts[env] = env
end
end
end)
return scripts
end
changeallenv=function(x,y)
table.foreach(getallenvironments(), function(z,v) pcall(function() rawset(v, x, y) end) end) end
oldreq=request
getgenv().request=function(x,y)
if getsettings().HttpEnabled and not x.Url:find("api.nexemulation") then
return oldreq(x,y)
end
warn("HTTP is disabled!")
return {Body="Disabled", Success=false, StatusCode=401, StatusMessage="Unauthorized",Headers={}, Cookies={}} 
end
